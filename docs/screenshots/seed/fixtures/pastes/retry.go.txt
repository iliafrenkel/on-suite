package retry

import (
	"context"
	"errors"
	"math/rand/v2"
	"time"
)

// ErrGiveUp wraps the last error once every attempt has failed.
var ErrGiveUp = errors.New("retry: giving up")

// Do calls fn until it succeeds, ctx is done, or attempts run out. The
// wait between tries doubles each time, with jitter so a fleet of clients
// does not retry in lockstep.
func Do(ctx context.Context, attempts int, base time.Duration, fn func(context.Context) error) error {
	var err error
	for i := range attempts {
		if err = fn(ctx); err == nil {
			return nil
		}
		if i == attempts-1 {
			break
		}
		wait := base << i
		wait += time.Duration(rand.Int64N(int64(wait) / 2))

		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(wait):
		}
	}
	return errors.Join(ErrGiveUp, err)
}
